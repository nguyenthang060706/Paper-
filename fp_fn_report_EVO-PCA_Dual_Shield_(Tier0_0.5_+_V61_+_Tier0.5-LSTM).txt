202	    if (check(TokenKind::LBracket)) {
203	        // Array destructuring: consume [ ... ] as raw text
204	        n->name = consumeDestructure(TokenKind::LBracket, TokenKind::RBracket, '[', ']');
205	        return n;
206	    }
207	
208	    n->name = expect(TokenKind::Identifier, "parameter name").value;
209	    // Default value: name = expr
210	    if (match(TokenKind::Equals)) {
211	        // Store " = <expr>" suffix in name for verbatim re-emit
212	        n->name += " = ";
213	        // Consume tokens until , or ) — simple greedy approach
214	        int depth = 0;
215	        while (!atEnd()) {
216	            TokenKind k = peek().kind;
217	            if (depth == 0 && (k == TokenKind::Comma || k == TokenKind::RParen)) break;
218	            if (k == TokenKind::LParen || k == TokenKind::LBrace || k == TokenKind::LBracket) ++depth;
219	            if (k == TokenKind::RParen || k == TokenKind::RBrace || k == TokenKind::RBracket) {
220	                if (depth == 0) break;
221	                --depth;
222	            }
223	            n->name += advance().value;
224	        }
225	    }
226	    return n;
227	}

229	// Consume a balanced bracket group and return it as a string.
230	// Uses inner spaces for object patterns "{ a, b }" but not array "[a, b]".
231	std::string Parser::consumeDestructure(TokenKind open, TokenKind close,
232	                                        char openCh, char closeCh) {
233	    expect(open, std::string(1, openCh));
234	    bool isObject = (openCh == '{');
235	    std::string s;
236	    s += openCh;
237	    if (isObject) s += ' ';
238	    int depth = 1;
239	    while (!atEnd() && depth > 0) {
240	        TokenKind k = peek().kind;
241	        if (k == open)  ++depth;
242	        if (k == close) { --depth; if (depth == 0) { advance(); break; } }
243	        s += advance().value;
244	        if (k == TokenKind::Comma) s += ' ';
245	    }
246	    if (isObject) s += ' ';
247	    s += closeCh;
248	    return s;
249	}

251	NodePtr Parser::parseFunctionDecl(bool isAsync) {
252	    const Token& start = expect(TokenKind::KW_function, "function");
253	    auto n = makeNode(NodeKind::FunctionDecl, start.line, start.col);
254	    n->async_ = isAsync;
255	    if (check(TokenKind::Identifier)) n->name = advance().value;
256	    expect(TokenKind::LParen, "(");
257	    while (!check(TokenKind::RParen) && !atEnd()) {
258	        n->params.push_back(parseParam());
259	        if (!match(TokenKind::Comma)) break;
260	    }
261	    expect(TokenKind::RParen, ")");
262	    n->body.push_back(parseBlockStmt());
263	    return n;
264	}

267	    const Token& start = expect(TokenKind::LBrace, "{");
268	    auto n = makeNode(NodeKind::BlockStmt, start.line, start.col);
269	    while (!check(TokenKind::RBrace) && !atEnd())
270	        n->body.push_back(parseStatement());
271	    expect(TokenKind::RBrace, "}");
272	    return n;
273	}

299	    const Token& start = expect(TokenKind::KW_for, "for");
300	    auto n = makeNode(NodeKind::ForStmt, start.line, start.col);
301	    expect(TokenKind::LParen, "(");
302	    if (!check(TokenKind::Semicolon))
303	        n->init = (check(TokenKind::KW_const) || check(TokenKind::KW_let) || check(TokenKind::KW_var))
304	                  ? parseVarDecl() : parseExprStmt();
305	    else match(TokenKind::Semicolon);
306	    if (!check(TokenKind::Semicolon)) n->test = parseExpr();
307	    expect(TokenKind::Semicolon, ";");
308	    if (!check(TokenKind::RParen))    n->update = parseExpr();
309	    expect(TokenKind::RParen, ")");
310	    n->body.push_back(parseStatement());
311	    return n;
312	}

324	NodePtr Parser::parseVarDecl() {
325	    const Token& start = peek();
326	    auto n = makeNode(NodeKind::VarDecl, start.line, start.col);
327	    n->varKind = advance().value; // const/let/var
328	    do {
329	        auto decl = makeNode(NodeKind::VarDeclarator, peek().line, peek().col);
330	        if (check(TokenKind::LBrace)) {
331	            decl->name = consumeDestructure(TokenKind::LBrace, TokenKind::RBrace, '{', '}');
332	        } else if (check(TokenKind::LBracket)) {
333	            decl->name = consumeDestructure(TokenKind::LBracket, TokenKind::RBracket, '[', ']');
334	        } else {
335	            decl->name = expect(TokenKind::Identifier, "variable name").value;
336	        }
337	        if (match(TokenKind::Equals)) {
338	            inExpressionContext_ = true;
339	            decl->init = parseAssignExpr();
340	            inExpressionContext_ = false;
341	        }
342	        n->declarations.push_back(std::move(decl));
343	    } while (match(TokenKind::Comma));
344	    match(TokenKind::Semicolon);
345	    return n;
346	}

349	    const Token& start = peek();
350	    auto n = makeNode(NodeKind::ExprStmt, start.line, start.col);
351	    n->expression = parseExpr();
352	    match(TokenKind::Semicolon);
353	    return n;
354	}

357	// Expression parsers (precedence climbing)
358	// ---------------------------------------------------------------------------
359	NodePtr Parser::parseExpr() {
360	    auto left = parseAssignExpr();
361	    if (check(TokenKind::Comma)) {
362	        auto seq = makeNode(NodeKind::SequenceExpr, left->line, left->col);
363	        seq->children.push_back(std::move(left));
364	        while (match(TokenKind::Comma))
365	            seq->children.push_back(parseAssignExpr());
366	        return seq;
367	    }
368	    return left;
369	}

371	NodePtr Parser::parseAssignExpr() {
372	    auto left = parseConditionalExpr();
373	
374	    TokenKind k = peek().kind;
375	    if (k == TokenKind::Equals       || k == TokenKind::PlusEquals  ||
376	        k == TokenKind::MinusEquals  || k == TokenKind::StarEquals   ||
377	        k == TokenKind::SlashEquals  || k == TokenKind::PercentEquals) {
378	        std::string op = advance().value;
379	        auto n = makeNode(NodeKind::AssignExpr, left->line, left->col);
380	        n->name  = op;
381	        n->object   = std::move(left);
382	        n->property = parseAssignExpr();
383	        return n;
384	    }
385	    return left;
386	}

388	NodePtr Parser::parseConditionalExpr() {
389	    auto cond = parseLogicalOrExpr();
390	    if (match(TokenKind::QuestionMark)) {
391	        auto n = makeNode(NodeKind::ConditionalExpr, cond->line, cond->col);
392	        n->test       = std::move(cond);
393	        n->consequent = parseAssignExpr();
394	        expect(TokenKind::Colon, ":");
395	        n->alternate  = parseAssignExpr();
396	        return n;
397	    }
398	    return cond;
399	}

401	static NodePtr makeBinary(NodeKind kind, NodePtr left, std::string op, NodePtr right) {
402	    auto n = makeNode(kind, left->line, left->col);
403	    n->name     = op;
404	    n->object   = std::move(left);
405	    n->property = std::move(right);
406	    return n;
407	}

410	    auto left = parseLogicalAndExpr();
411	    while (check(TokenKind::PipePipe) || check(TokenKind::QuestionQuestion)) {
412	        std::string op = advance().value;
413	        left = makeBinary(NodeKind::LogicalExpr, std::move(left), op, parseLogicalAndExpr());
414	    }
415	    return left;
416	}

468	           check(TokenKind::KW_in)        || check(TokenKind::KW_instanceof)) {

478	        std::string op = advance().value;

496	        std::string op = advance().value;
497	        left = makeBinary(NodeKind::BinaryExpr, std::move(left), op, parseUnaryExpr());
498	    }

502	NodePtr Parser::parseUnaryExpr() {
503	    if (check(TokenKind::Bang)      || check(TokenKind::Minus)      ||
504	        check(TokenKind::Plus)      || check(TokenKind::Tilde)      ||
505	        check(TokenKind::KW_typeof) || check(TokenKind::KW_void)    ||
506	        check(TokenKind::KW_delete) || check(TokenKind::KW_await)) {
507	        const Token& op = advance();
508	        auto n = makeNode(NodeKind::UnaryExpr, op.line, op.col);
509	        n->name   = op.value;
510	        n->object = parseUnaryExpr();
511	        return n;
512	    }
513	    return parseUpdateExpr();
514	}

516	NodePtr Parser::parseUpdateExpr() {
517	    if (check(TokenKind::PlusPlus) || check(TokenKind::MinusMinus)) {
518	        const Token& op = advance();
519	        auto n = makeNode(NodeKind::UnaryExpr, op.line, op.col);
520	        n->name   = op.value + "pre";
521	        n->object = parseCallMemberExpr();
522	        return n;
523	    }
524	    auto expr = parseCallMemberExpr();
525	    if (check(TokenKind::PlusPlus) || check(TokenKind::MinusMinus)) {
526	        const Token& op = advance();
527	        auto n = makeNode(NodeKind::UnaryExpr, op.line, op.col);
528	        n->name   = op.value + "post";
529	        n->object = std::move(expr);
530	        return n;
531	    }
532	    return expr;
533	}

535	NodePtr Parser::parseCallMemberExpr() {
536	    auto expr = parsePrimaryExpr();
537	
538	    while (true) {
539	        if (match(TokenKind::LParen)) {
540	            // call
541	            auto call = makeNode(NodeKind::CallExpr, expr->line, expr->col);
542	            call->callee = std::move(expr);
543	            while (!check(TokenKind::RParen) && !atEnd()) {
544	                if (match(TokenKind::DotDotDot)) {
545	                    auto sp = makeNode(NodeKind::SpreadElement, peek().line, peek().col);
546	                    sp->object = parseAssignExpr();
547	                    call->arguments.push_back(std::move(sp));
548	                } else {
549	                    call->arguments.push_back(parseAssignExpr());
550	                }
551	                if (!match(TokenKind::Comma)) break;
552	            }
553	            expect(TokenKind::RParen, ")");
554	            expr = std::move(call);
555	        } else if (match(TokenKind::Dot)) {
556	            auto mem = makeNode(NodeKind::MemberExpr, expr->line, expr->col);
557	            mem->object   = std::move(expr);
558	            auto prop     = makeNode(NodeKind::Identifier, peek().line, peek().col);
559	            prop->name    = advance().value;
560	            mem->property = std::move(prop);
561	            mem->computed = false;
562	            expr = std::move(mem);
563	        } else if (match(TokenKind::LBracket)) {
564	            auto mem = makeNode(NodeKind::MemberExpr, expr->line, expr->col);
565	            mem->object   = std::move(expr);
566	            mem->property = parseExpr();
567	            mem->computed = true;
568	            expect(TokenKind::RBracket, "]");
569	            expr = std::move(mem);
570	        } else {
571	            break;
572	        }
573	    }
574	    return expr;
575	}

577	NodePtr Parser::parsePrimaryExpr() {
578	    const Token& t = peek();
579	
580	    // JSX — triggered by JSXLess token (lexer already disambiguated)
581	    if (t.kind == TokenKind::JSXLess) {
582	        if (peek(1).kind == TokenKind::JSXGreater) // <>
583	            return parseJSXFragment();
584	        return parseJSXElement();
585	    }
586	
587	    switch (t.kind) {
588	        case TokenKind::Number: {
589	            auto n = makeNode(NodeKind::NumericLiteral, t.line, t.col);
590	            n->value = advance().value;
591	            return n;
592	        }
593	        case TokenKind::String: {
594	            auto n = makeNode(NodeKind::StringLiteral, t.line, t.col);
595	            n->value = advance().value;
596	            return n;
597	        }
598	        case TokenKind::TemplateString: {
599	            return parseTemplateLiteral();
600	        }
601	        case TokenKind::KW_true:
602	        case TokenKind::KW_false: {
603	            auto n = makeNode(NodeKind::BoolLiteral, t.line, t.col);
604	            n->boolVal = (advance().kind == TokenKind::KW_true);
605	            return n;
606	        }
607	        case TokenKind::KW_null: {
608	            advance();
609	            return makeNode(NodeKind::NullLiteral, t.line, t.col);
610	        }
611	        case TokenKind::KW_this: {
612	            auto n = makeNode(NodeKind::Identifier, t.line, t.col);
613	            n->name = advance().value;
614	            return n;
615	        }
616	        case TokenKind::Identifier: {
617	            auto n = makeNode(NodeKind::Identifier, t.line, t.col);
618	            n->name = advance().value;
619	            // Arrow: (a) => ... or a => ...
620	            if (check(TokenKind::Arrow)) {
621	                advance(); // =>
622	                auto arrow = makeNode(NodeKind::ArrowFunctionExpr, n->line, n->col);
623	                auto param = makeNode(NodeKind::Identifier, n->line, n->col);
624	                param->name = n->name;
625	                arrow->params.push_back(std::move(param));
626	                if (check(TokenKind::LBrace))
627	                    arrow->body.push_back(parseBlockStmt());
628	                else {
629	                    inExpressionContext_ = true;
630	                    arrow->body.push_back(parseAssignExpr());
631	                    inExpressionContext_ = false;
632	                }
633	                return arrow;
634	            }
635	            return n;
636	        }
637	        case TokenKind::LParen: {
638	            advance(); // (
639	            // Could be grouping (expr) or arrow params (p1, p2) =>
640	            if (check(TokenKind::RParen)) {
641	                // () =>
642	                advance();
643	                expect(TokenKind::Arrow, "=>");
644	                auto arrow = makeNode(NodeKind::ArrowFunctionExpr, t.line, t.col);
645	                if (check(TokenKind::LBrace))
646	                    arrow->body.push_back(parseBlockStmt());
647	                else {
648	                    inExpressionContext_ = true;
649	                    arrow->body.push_back(parseAssignExpr());
650	                    inExpressionContext_ = false;
651	                }
652	                return arrow;
653	            }
654	            // If the first token is { or [ or ... it must be arrow params —
655	            // parse all params eagerly so destructuring is handled correctly.
656	            if (check(TokenKind::LBrace) || check(TokenKind::LBracket) ||
657	                check(TokenKind::DotDotDot)) {
658	                auto arrow = makeNode(NodeKind::ArrowFunctionExpr, t.line, t.col);
659	                while (!check(TokenKind::RParen) && !atEnd()) {
660	                    arrow->params.push_back(parseParam());
661	                    if (!match(TokenKind::Comma)) break;
662	                }
663	                expect(TokenKind::RParen, ")");
664	                expect(TokenKind::Arrow, "=>");
665	                if (check(TokenKind::LBrace))
666	                    arrow->body.push_back(parseBlockStmt());
667	                else {
668	                    inExpressionContext_ = true;
669	                    arrow->body.push_back(parseAssignExpr());
670	                    inExpressionContext_ = false;
671	                }
672	                return arrow;
673	            }
674	            inExpressionContext_ = true;
675	            auto inner = parseExpr();
676	            inExpressionContext_ = false;
677	            expect(TokenKind::RParen, ")");
678	            if (check(TokenKind::Arrow)) {
679	                // treat as arrow with single identifier/grouped param
680	                advance();
681	                auto arrow = makeNode(NodeKind::ArrowFunctionExpr, t.line, t.col);
682	                arrow->params.push_back(std::move(inner));
683	                if (check(TokenKind::LBrace))
684	                    arrow->body.push_back(parseBlockStmt());
685	                else {
686	                    inExpressionContext_ = true;
687	                    arrow->body.push_back(parseAssignExpr());
688	                    inExpressionContext_ = false;
689	                }
690	                return arrow;
691	            }
692	            return inner;
693	        }
694	        case TokenKind::LBracket: return parseArrayExpr();
695	        case TokenKind::LBrace:   return parseObjectExpr();
696	        case TokenKind::KW_function: return parseFunctionExpr(false);
697	        case TokenKind::KW_async: {
698	            if (peek(1).kind == TokenKind::KW_function) {
699	                advance();
700	                return parseFunctionExpr(true);
701	            }
702	            // async arrow: async (params) => body
703	            advance();
704	            auto arrow = makeNode(NodeKind::ArrowFunctionExpr, t.line, t.col);
705	            arrow->async_ = true;
706	            if (check(TokenKind::LParen)) {
707	                advance();
708	                while (!check(TokenKind::RParen) && !atEnd()) {
709	                    arrow->params.push_back(parseParam());
710	                    if (!match(TokenKind::Comma)) break;
711	                }
712	                expect(TokenKind::RParen, ")");
713	            } else {
714	                arrow->params.push_back(parseParam());
715	            }
716	            expect(TokenKind::Arrow, "=>");
717	            if (check(TokenKind::LBrace))
718	                arrow->body.push_back(parseBlockStmt());
719	            else {
720	                inExpressionContext_ = true;
721	                arrow->body.push_back(parseAssignExpr());
722	                inExpressionContext_ = false;
723	            }
724	            return arrow;
725	        }
726	        case TokenKind::KW_new: {
727	            advance();
728	            auto n = makeNode(NodeKind::NewExpr, t.line, t.col);
729	            n->callee = parseCallMemberExpr();
730	            if (match(TokenKind::LParen)) {
731	                while (!check(TokenKind::RParen) && !atEnd()) {
732	                    n->arguments.push_back(parseAssignExpr());
733	                    if (!match(TokenKind::Comma)) break;
734	                }
735	                expect(TokenKind::RParen, ")");
736	            }
737	            return n;
738	        }
739	        default:
740	            error("Unexpected token '" + t.value + "'");
741	            return nullptr; // unreachable
742	    }
743	}

745	NodePtr Parser::parseFunctionExpr(bool isAsync) {
746	    const Token& start = expect(TokenKind::KW_function, "function");
747	    auto n = makeNode(NodeKind::FunctionDecl, start.line, start.col);
748	    n->async_ = isAsync;
749	    if (check(TokenKind::Identifier)) n->name = advance().value;
750	    expect(TokenKind::LParen, "(");
751	    while (!check(TokenKind::RParen) && !atEnd()) {
752	        n->params.push_back(parseParam());
753	        if (!match(TokenKind::Comma)) break;
754	    }
755	    expect(TokenKind::RParen, ")");
756	    n->body.push_back(parseBlockStmt());
757	    return n;
758	}

760	NodePtr Parser::parseArrayExpr() {
761	    const Token& start = expect(TokenKind::LBracket, "[");
762	    auto n = makeNode(NodeKind::ArrayExpr, start.line, start.col);
763	    while (!check(TokenKind::RBracket) && !atEnd()) {
764	        if (match(TokenKind::DotDotDot)) {
765	            auto sp = makeNode(NodeKind::SpreadElement, peek().line, peek().col);
766	            sp->object = parseAssignExpr();
767	            n->children.push_back(std::move(sp));
768	        } else {
769	            n->children.push_back(parseAssignExpr());
770	        }
771	        if (!match(TokenKind::Comma)) break;
772	    }
773	    expect(TokenKind::RBracket, "]");
774	    return n;
775	}

777	NodePtr Parser::parseObjectExpr() {
778	    const Token& start = expect(TokenKind::LBrace, "{");
779	    auto n = makeNode(NodeKind::ObjectExpr, start.line, start.col);
780	    while (!check(TokenKind::RBrace) && !atEnd()) {
781	        auto prop = makeNode(NodeKind::ObjectProperty, peek().line, peek().col);
782	        if (match(TokenKind::DotDotDot)) {
783	            prop->name = "...";
784	            prop->init  = parseAssignExpr();
785	        } else {

826	NodePtr Parser::parseJSXElement() {
827	    const Token& lt = expect(TokenKind::JSXLess, "<");
828	
829	    // Check if the very next thing is a closing tag — shouldn't happen at top
830	    // level but guard anyway.
831	    std::string tagName = parseJSXTagName();
832	
833	    auto opening = makeNode(NodeKind::JSXOpeningElement, lt.line, lt.col);
834	    opening->name = tagName;
835	
836	    // Parse attributes
837	    while (!check(TokenKind::JSXGreater) && !check(TokenKind::JSXSlashGreater) && !atEnd()) {
838	        opening->attributes.push_back(parseJSXAttribute());
839	    }
840	
841	    if (match(TokenKind::JSXSlashGreater)) {
842	        // Self-closing <Tag />
843	        opening->selfClosing = true;
844	        auto el = makeNode(NodeKind::JSXElement, lt.line, lt.col);
845	        el->openingElement = std::move(opening);
846	        el->selfClosing    = true;
847	        return el;
848	    }
849	
850	    expect(TokenKind::JSXGreater, ">");
851	
852	    // Children
853	    auto el = makeNode(NodeKind::JSXElement, lt.line, lt.col);
854	    el->openingElement = std::move(opening);
855	    el->children       = parseJSXChildren();
856	
857	    // Closing tag
858	    auto closing = parseJSXClosingElement();
859	    if (closing->name != tagName) {
860	        throw ParseError("Mismatched JSX tags: <" + tagName + "> closed by </" + closing->name + ">",
861	                         closing->line, closing->col);
862	    }
863	    el->closingElement = std::move(closing);
864	    return el;
865	}

867	NodePtr Parser::parseJSXFragment() {
868	    const Token& lt = expect(TokenKind::JSXLess, "<");
869	    expect(TokenKind::JSXGreater, ">");

875	    expect(TokenKind::JSXLessSlash, "</");

880	NodeList Parser::parseJSXChildren() {
881	    NodeList children;
882	    while (!atEnd()) {
883	        if (check(TokenKind::JSXLessSlash)) break; // </Tag>
884	
885	        if (check(TokenKind::JSXText)) {
886	            const Token& t = advance();
887	            // Skip whitespace-only text nodes
888	            bool allSpace = true;
889	            for (char c : t.value) if (!std::isspace((unsigned char)c)) { allSpace = false; break; }
890	            if (!allSpace) {
891	                auto txt = makeNode(NodeKind::JSXText, t.line, t.col);
892	                txt->value = t.value;
893	                children.push_back(std::move(txt));
894	            }
895	        } else if (check(TokenKind::LBrace)) {
896	            children.push_back(parseJSXExprContainer());
897	        } else if (check(TokenKind::JSXLess)) {
898	            // Nested element or fragment
899	            if (peek(1).kind == TokenKind::JSXGreater)
900	                children.push_back(parseJSXFragment());
901	            else
902	                children.push_back(parseJSXElement());
903	        } else {
904	            break;
905	        }
906	    }
907	    return children;
908	}

912	    auto n = makeNode(NodeKind::JSXClosingElement, ls.line, ls.col);
913	    n->name = parseJSXTagName();
914	    expect(TokenKind::JSXGreater, ">");
915	    return n;

918	NodePtr Parser::parseJSXAttribute() {
919	    // Spread: {...obj}
920	    if (check(TokenKind::LBrace)) {
921	        advance();
922	        expect(TokenKind::DotDotDot, "...");
923	        auto n = makeNode(NodeKind::JSXSpreadAttribute, peek().line, peek().col);
924	        n->expression = parseAssignExpr();
925	        expect(TokenKind::RBrace, "}");
926	        return n;
927	    }
928	
929	    auto n = makeNode(NodeKind::JSXAttribute, peek().line, peek().col);
930	    n->name = expect(TokenKind::Identifier, "attribute name").value;
931	    // Optional: attr-name can contain '-' (data-x) — handled as separate tokens;
932	    // for MVP we only handle simple identifiers.
933	
934	    if (!match(TokenKind::Equals)) {
935	        // Boolean attribute: <Tag disabled />
936	        auto bval = makeNode(NodeKind::BoolLiteral, n->line, n->col);
937	        bval->boolVal = true;
938	        n->init = std::move(bval);
939	        return n;
940	    }
941	
942	    if (check(TokenKind::JSXString)) {
943	        auto sv = makeNode(NodeKind::StringLiteral, peek().line, peek().col);
944	        sv->value = advance().value;
945	        n->init = std::move(sv);
946	    } else if (check(TokenKind::LBrace)) {
947	        n->init = parseJSXExprContainer();
948	    } else {
949	        error("Expected attribute value");
950	    }
951	    return n;
952	}

954	NodePtr Parser::parseJSXExprContainer() {
955	    const Token& lb = expect(TokenKind::LBrace, "{");
956	    auto n = makeNode(NodeKind::JSXExpressionContainer, lb.line, lb.col);
957	    inExpressionContext_ = true;
958	    n->expression = parseAssignExpr();
959	    inExpressionContext_ = false;
960	    expect(TokenKind::RBrace, "}");
961	    return n;
962	}
